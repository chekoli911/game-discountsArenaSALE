  function alphaSortKey(item){
    // A–Я: 007 First Light → латиница → кириллица → цифры/прочее
    if(is007FirstLight(item)) return '0\t007 first light';
    const raw=String((item&&item.title)||'').trim();
    const t=raw.toLowerCase().replace(/ё/g,'е');
    let ch='';
    for(let i=0;i<t.length;i++){
      const c=t.charAt(i);
      if(/[a-zа-я0-9]/i.test(c)){ ch=c; break; }
    }
    if(!ch) ch=t.charAt(0)||'~';
    const code=ch.charCodeAt(0);
    let group=3;
    if(code>=97&&code<=122) group=1;
    else if(code>=1072&&code<=1103) group=2;
    else group=3;
    return group+'\t'+t;
  }

      if(searchIndexMatches(row, needles)) want.add(+row.chunk||0);
    }
    if(!want.size) return;
    const jobs=[];
    for(const idx of want) jobs.push(loadChunkByIndex(idx).catch(()=>[]));
    await Promise.all(jobs);
  }

  async function ensureChunksForPage(page){
    const chunks=(state.manifest&&state.manifest.chunks)||[];
    if(!chunks.length) return false;
    const chunkSize=Math.max(1, +(state.manifest&&state.manifest.chunk_size)||100);
    const needItems=(page+1)*pageSize();
    const needChunk=Math.min(chunks.length-1, Math.floor(Math.max(0, needItems-1)/chunkSize));
    const jobs=[];
    for(let i=0;i<=needChunk;i++){
      if(!state.chunksLoaded.has(i)) jobs.push(loadChunkByIndex(i));
    }
    if(!jobs.length) return false;
    await Promise.all(jobs.map(p=>p.catch(()=>{})));
    return true;
  }
  async function loadAllChunksBackground(manifest, opts){
    opts=opts||{};
    const chunks=(manifest&&manifest.chunks)||[];
    const batch=8;
    const startIdx=opts.fromIdx!=null?opts.fromIdx:1;
    for(let i=startIdx;i<chunks.length;i+=batch){
      const jobs=[];
      for(let j=i;j<Math.min(i+batch, chunks.length);j++){
        jobs.push(loadChunk(chunks[j], j).catch(()=>{}));
      }
      await Promise.all(jobs);
      setCatalogLoadingStatus(state.chunksLoaded.size, chunks.length);
      // Не перерисовываем первую страницу при каждой догрузке — карточки не прыгают
      if(state.page>0 || state.filter!=='all' || String(state.query||'').trim()){
        if(!state.open) scheduleRenderGrid(false);
      }
      await new Promise(r=>setTimeout(r,10));
    }
    state.allChunksLoaded=true;
    state.firstPageReady=true;
    invalidateSpotlight();
    clearCatalogLoadingTip();
    $('[data-filters]',mount).innerHTML=filterChips();
    if(!state.open) requestAnimationFrame(()=>renderGrid());
  }

  function animatePriceTo(target, instant){
    const el=$('[data-price]', mount);
    if(!el){ state.priceShown=target; return; }
    const to=Math.round(+target||0);
    if(instant||state.priceShown==null){
      if(state.priceAnimId) cancelAnimationFrame(state.priceAnimId);
      state.priceShown=to; el.textContent=nf(to); return;
    }
    const from=Math.round(+state.priceShown||0);
    if(from===to){ el.textContent=nf(to); state.priceShown=to; return; }
    if(state.priceAnimId) cancelAnimationFrame(state.priceAnimId);
    const start=performance.now(), dur=480;
    const tick=now=>{
      const t=Math.min(1,(now-start)/dur);
      const eased=1-Math.pow(1-t,3);
      const val=Math.round(from+(to-from)*eased);
      el.textContent=nf(val);
      if(t<1) state.priceAnimId=requestAnimationFrame(tick);
      else{ state.priceShown=to; el.textContent=nf(to); }
    };
    state.priceAnimId=requestAnimationFrame(tick);
  }

  function filteredItems(){
    let list=state.items.filter(x=>x&&!isHiddenRentItem(x));
    const needles=searchNeedles(state.query);
    if(needles.length){
      list=list.filter(g=>gameMatchesQuery(g,needles));
    }
    if(state.filter==='new'){
      if(!state.novinkiReady){
        return [];
      }
      // Список с PS Store «New games» (rent_novinki.json)
      list=list.filter(isNovinkiItem);
      const orderList=state.novinkiOrder;
      if(orderList && orderList.length){
        const order=new Map(orderList.map((s,i)=>[s,i]));
        list=list.slice().sort((a,b)=>(order.get(a.slug)??9999)-(order.get(b.slug)??9999));
      }
    } else if(state.filter==='recent'){
      // Только свежие карточки из синка discounts/vserent (catalog_added_at), не весь каталог
      list=list.filter(isRecentlyAddedItem).sort((a,b)=>recencyKey(b)-recencyKey(a));
    } else if(state.filter==='az'){
      list=list.slice().sort((a,b)=>alphaSortKey(a).localeCompare(alphaSortKey(b),'en'));
    } else if(!needles.length){
      // ВСЕ без поиска: стр.1–2 = будущие (+актуальные), дальше остальные по дате
      list=getHomeCatalogList(list);
      // На 3-й странице дубли «Актуальных» убираем только из хвоста — 1–2 страницы не съезжают
      if(showSpotlight() && state.allChunksLoaded){
        const frontN=pageSize()*2;
        const head=list.slice(0, frontN);
        const skip=new Set(getSpotlightItems().map(x=>x.slug));
        const tail=list.slice(frontN).filter(x=>!skip.has(x.slug));
        list=head.concat(tail);
      }
    } else {
      list=list.slice().sort((a,b)=>releaseSortKey(b)-releaseSortKey(a));
    }
    return list;
  }

  function syncCardList(parent, items, opts){

  function renderGrid(){
    const grid=$('[data-grid]',mount);
    if(!grid) return;
    // Ждём только первый чанк — страница 1 рисуется сразу, остальное догружается в фоне
    if(!state.firstPageReady){
      const total=((state.manifest&&state.manifest.chunks)||[]).length||0;
      setCatalogLoadingStatus(state.chunksLoaded.size, total);
      renderPageMeta(0, 0);
      const spot=$('[data-spotlight]',mount);
      if(spot){ spot.innerHTML=''; spot.hidden=true; }
      return;
    }
    renderSpotlight();
    if(state.filter==='new' && !state.novinkiReady){
      grid.innerHTML='<div class="rt-status">'+(state.novinkiLoading?'Загрузка новинок…':'Новинки временно недоступны')+'</div>';
      renderPageMeta(0, 0);
      return;
    }
    const list=filteredItems();
    if(!list.length && !showSpotlight()){
      if(!state.allChunksLoaded && (state.filter!=='all' || String(state.query||'').trim())){
        grid.innerHTML='<div class="rt-status">Загрузка каталога…</div>';
        renderPageMeta(0, 0);
        return;
      }
      grid.innerHTML='<div class="rt-empty">Ничего не найдено</div>';
      renderPageMeta(0, 0);
      grid.classList.add('is-ready');
      return;
    }
    if(!list.length && showSpotlight() && getSpotlightItems().length){
      grid.querySelectorAll('[data-open-slug],.rt-pager,.rt-pager__info,.rt-empty,.rt-status').forEach(el=>el.remove());
      renderPageMeta(0, 1);
      grid.classList.add('is-ready');
      return;
    }
    const useManifestPages=state.filter==='all' && !String(state.query||'').trim();
    const totalForPages=useManifestPages ? Math.max(list.length, catalogTotalEstimate()) : list.length;
    const pages=Math.max(1, Math.ceil(totalForPages/pageSize()));
    if(state.page>pages-1) state.page=pages-1;
    if(state.page<0) state.page=0;
    const needFrom=state.page*pageSize();
    const needTo=needFrom+pageSize();
    if(!state.allChunksLoaded && list.length < needTo){
      const chunkSize=Math.max(1, +(state.manifest&&state.manifest.chunk_size)||100);
      const chunks=(state.manifest&&state.manifest.chunks)||[];
      const needChunk=chunks.length
        ? Math.min(chunks.length-1, Math.floor(Math.max(0, needTo-1)/chunkSize))
        : 0;
      let missing=0;
      for(let i=0;i<=needChunk;i++){
        if(!state.chunksLoaded.has(i)) missing++;
      }
      if(missing && !state.ensuringPage){
        grid.querySelector('.rt-empty')?.remove();
        let st=grid.querySelector('.rt-status');
        if(!st){
          st=document.createElement('div');
          st.className='rt-status';
          grid.appendChild(st);
        }
        st.textContent='Загрузка страницы…';
        state.ensuringPage=true;
        ensureChunksForPage(state.page).finally(()=>{
          state.ensuringPage=false;
          if(!state.open) renderGrid();
        });
        updatePager(grid, state.page, pages, 0);
        renderPageMeta(0, pages);
        return;
      }
      // Чанки уже есть (или грузятся) — рисуем то, что успело, без бесконечного цикла
    }
    const slice=list.slice(needFrom, needTo);
    grid.querySelector('.rt-empty,.rt-status')?.remove();
    syncCardList(grid, slice, {extraClass:''});
    updatePager(grid, state.page, pages, slice.length);
    renderPageMeta(slice.length, pages);
    grid.classList.add('is-ready');
  }

  function pagerHTML(page, pages, gridCount){
    const cur=page+1;
    const info=pages>1?`<div class="rt-pager__info">Страница ${cur} из ${pages}</div>`:'';
    if(pages<=1) return info;
    const nums=[];
    const push=(n)=>nums.push(n);
    if(pages<=12){
      for(let i=1;i<=pages;i++) push(i);
    }else{
      push(1);
      let start=Math.max(2, cur-2);
      let end=Math.min(pages-1, cur+2);
      if(cur<=4){ start=2; end=6; }
      if(cur>=pages-3){ start=pages-5; end=pages-1; }
      if(start>2) nums.push('…');
      for(let i=start;i<=end;i++) push(i);
      if(end<pages-1) nums.push('…');
      push(pages);
    }
    const btns=nums.map(n=>{
      if(n==='…') return `<span class="rt-pager__dots">…</span>`;
      return `<button type="button" class="rt-pager__btn${n===cur?' is-on':''}" data-page="${n-1}">${n}</button>`;
    }).join('');
    return `<div class="rt-pager">
      <button type="button" class="rt-pager__btn" data-page="${Math.max(0,page-1)}" ${page<=0?'disabled':''} aria-label="Назад">‹</button>
      ${btns}
      <button type="button" class="rt-pager__btn" data-page="${Math.min(pages-1,page+1)}" ${page>=pages-1?'disabled':''} aria-label="Вперёд">›</button>
    </div>${info}`;
  }

  function filterChips(){
    const chips=[
      {id:'all',label:'ВСЕ'},
      {id:'new',label:'НОВИНКИ'},
      {id:'recent',label:'Недавно добавленные'},
      {id:'az',label:'A–Я'}
    ].map(c=>`<button type="button" class="rt-chip${state.filter===c.id?' is-on':''}" data-filter="${attr(c.id)}">${esc(c.label)}</button>`).join('');
    const gridIcon=(cols, cells)=>{
      const size=16, pad=1.2, gap=1.4;
      const colsN=cols, rows=2;
      const cw=(size-pad*2-gap*(colsN-1))/colsN;
      const rh=(size-pad*2-gap*(rows-1))/rows;
      let rects='';
      for(let r=0;r<rows;r++){
        for(let c=0;c<colsN;c++){
          const x=pad+c*(cw+gap), y=pad+r*(rh+gap);
          rects+=`<rect x="${x.toFixed(2)}" y="${y.toFixed(2)}" width="${cw.toFixed(2)}" height="${rh.toFixed(2)}" rx="1"/>`;

  function recommendations(excludeSlug){
    const hidden = (x)=>!x || !x.slug || x.slug===excludeSlug || isHiddenRentItem(x);
    const need = REC_MAX;
    const pri = state.recsPriority || {};
    const priSlugs = Array.isArray(pri.slugs) ? pri.slugs : [];
    const novSlugs = new Set(Array.isArray(pri.novinka_slugs) ? pri.novinka_slugs : []);
    const priSet = new Set(priSlugs);

    const isNovinkaItem = (x)=>{
      if(!x) return false;
      if(novSlugs.has(x.slug)) return true;
      if(x.is_new || isPreorderItem(x)) return true;
      const d=parseReleaseDate(x.release_date);
      if(!d) return false;
      const age=(Date.now()-d.getTime())/(24*3600*1000);
      return age>=0 && age<=120;
    };

    const vserentPool = priSlugs.map(s=>state.itemsBySlug[s]).filter(x=>!hidden(x));
    const novPool = shuffleArray(vserentPool.filter(isNovinkaItem));
    const vserentOther = shuffleArray(vserentPool.filter(x=>!isNovinkaItem(x)));

    const needVs = Math.max(0, Math.round(need * REC_VSERENT_SHARE));
    const needNov = Math.max(0, Math.ceil(needVs * REC_NOVINKA_IN_VSERENT));

    const picked=[];
    const used=new Set();
    const take=(pool, n)=>{
      for(let i=0;i<pool.length && n>0 && picked.length<need;i++){
        const it=pool[i];
        if(used.has(it.slug)) continue;
        used.add(it.slug);
        picked.push(it);
        n--;
      }
    };

    take(novPool, needNov);
    take(vserentOther, needVs - picked.length);
    take(novPool, needVs - picked.length);

    const otherPool = shuffleArray(
      state.items.filter(x=>!hidden(x) && !used.has(x.slug) && !priSet.has(x.slug))
    );
    take(otherPool, need - picked.length);

    if(picked.length < need){
      const any = shuffleArray(state.items.filter(x=>!hidden(x) && !used.has(x.slug)));
      take(any, need - picked.length);
    }
    return shuffleArray(picked);
  }

  async function ensureRecsPriority(){
    const now=Date.now();
    if(state.recsPriority && (now - (state.recsPriorityLoadedAt||0)) < REC_PRIORITY_TTL_MS){
      return state.recsPriority;
    }
    try{
      const cached=JSON.parse(localStorage.getItem(REC_PRIORITY_CACHE_KEY)||'null');
      if(cached && cached.data && cached.ts && (now-cached.ts) < REC_PRIORITY_TTL_MS){
        state.recsPriority=cached.data;
        state.recsPriorityLoadedAt=cached.ts;
        return state.recsPriority;
      }
    }catch(e){}
    try{
      const data=await fetchJson(RECS_PRIORITY_URLS);
      if(data && Array.isArray(data.slugs)){
        state.recsPriority=data;
        state.recsPriorityLoadedAt=now;
        try{ localStorage.setItem(REC_PRIORITY_CACHE_KEY, JSON.stringify({ts:now, data})); }catch(e){}
      }
    }catch(e){
      console.warn('Rent: recs priority load failed', e);
    }
    return state.recsPriority;
  }

  async function fetchNovinkiDoc(){
    let best=null;
    let lastErr=null;
    for(const url of NOVINKI_URLS){
      try{
        const r=await fetch(url+(url.indexOf('?')>=0?'&':'?')+'t='+Date.now(),{cache:'no-store'});
        if(!r.ok) throw new Error(r.statusText);
        const data=await r.json();
        const gen=String(data&&data.generated_at||'');
        if(!best || gen > String(best.generated_at||'')) best=data;
      }catch(e){ lastErr=e; }
    }
    if(!best) throw lastErr||new Error('novinki fetch failed');
    return best;
  }

  function applyNovinkiDoc(data, now){
    const slugs=Array.isArray(data&&data.slugs)?data.slugs.filter(Boolean):[];
    state.novinkiOrder=slugs;
    state.novinkiSlugs=new Set(slugs);
    state.novinkiGeneratedAt=String(data&&data.generated_at||'');

  async function ensureNovinki(){
    const now=Date.now();
    if(state.novinkiReady && state.novinkiOrder && state.novinkiOrder.length && (now - (state.novinkiLoadedAt||0)) < REC_PRIORITY_TTL_MS){
      return state.novinkiOrder;
    }
    if(state.novinkiLoading) return state.novinkiOrder;
    state.novinkiLoading=true;
    try{
      const data=await fetchNovinkiDoc();
      applyNovinkiDoc(data, now);
    }catch(e){
      console.warn('Rent: novinki load failed', e);
      if(state.novinkiOrder && state.novinkiOrder.length){
        state.novinkiReady=true;
      }else{
        state.novinkiOrder=[];
        state.novinkiSlugs=new Set();
      }
    }finally{
      state.novinkiLoading=false;
    }
    return state.novinkiOrder;
  }

    if(updateHistory){
      try{
        if(historyMode==='push') history.pushState({rentCard:true,pushed:true,slug:slug},'',newHash);
        else history.replaceState({rentCard:true,pushed:false,slug:slug},'',newHash);
      }catch(e){}
    }

    const card=state.itemsBySlug[slug]||state.items.find(x=>x.slug===slug);
    if(!card){
      state.pendingOpenSlug=slug;
      state.pendingOpenHistory=historyMode;

        state.allChunksLoading=true;

    paint();
    ensureRecsPriority().then(()=>{

        state.recsForSlug='';
        paint();
      }
    });
    $('#rtDetail',mount).classList.add('is-open');

    document.body.classList.add('rt-lock');
    const sc=$('.rt-detail__scroll',mount); if(sc) sc.scrollTop=0;

  function stepShot(delta){
    if(!state.open||!state.card) return false;
    const shots=shotsOf(state.card);
    if(shots.length<=1) return false;
    const next=(state.shotIdx+delta+shots.length)%shots.length;
    if(next===state.shotIdx) return false;
    state.shotIdx=next;
    if(isGalleryOpen()){
      refreshGalleryModal();

      if(inner){
        const hero=$('.rt-hero img',inner);
        if(hero) hero.src=shots[state.shotIdx]||shots[0];
        const count=$('.rt-gallery-count',inner);
        if(count) count.textContent=`${state.shotIdx+1} / ${shots.length}`;
        $$('[data-shot]',inner).forEach(b=>{
          const i=+(b.getAttribute('data-shot')||0);
          b.classList.toggle('is-on', i===state.shotIdx);
        });
      }
    }else renderDetail({animatePrice:true});
    return true;
  }
  function bindGallerySwipe(modal){
    if(!modal||modal.dataset.swipeBound) return;
    modal.dataset.swipeBound='1';
    let x0=null,y0=null,active=false;
    const start=(e)=>{
      const t=e.changedTouches?e.changedTouches[0]:null;
      if(!t) return;
      x0=t.clientX; y0=t.clientY; active=true;
    };
    const end=(e)=>{
      if(!active||x0==null) return;
      const t=e.changedTouches?e.changedTouches[0]:null;
      active=false;
      if(!t) return;
      const dx=t.clientX-x0, dy=t.clientY-y0;
      x0=null; y0=null;
      if(Math.abs(dx)<48 || Math.abs(dx)<Math.abs(dy)*1.2) return;
      if(dx<0) stepShot(1);
      else stepShot(-1);
    };

  function bindUI(){
    mount.addEventListener('change', e=>{
      const sel=e.target.closest('[data-page-size]');
      if(!sel||!mount.contains(sel)) return;
      setPageSize(sel.value);
      $('[data-filters]',mount).innerHTML=filterChips();
      renderGrid();
      scrollCatalogToTop();
    });
    mount.addEventListener('click', e=>{
      const t=e.target.closest('[data-open-slug],[data-filter],[data-theme-toggle],[data-close-detail],[data-sticky-buy],[data-sticky-cart],[data-sticky-share],[data-close-gallery],[data-gallery-prev],[data-gallery-next],[data-page],[data-fab-up],[data-search-clear],[data-grid-cols]');
      if(!t||!mount.contains(t)) return;
      if(t.hasAttribute('data-search-clear')){
        state.query='';
        state.page=0;
        $$('[data-top-search]',mount).forEach(x=>{ x.value=''; });
        syncSearchClear();
        if(state.open) closeDetail();
        $('[data-filters]',mount).innerHTML=filterChips();
        renderGrid();
        return;
      }
      if(t.hasAttribute('data-grid-cols')){
        setGridCols(t.getAttribute('data-grid-cols'));
        $('[data-filters]',mount).innerHTML=filterChips();
        return;
      }
      if(t.hasAttribute('data-open-slug')){
        openDetailBySlug(t.getAttribute('data-open-slug'), {history: t.getAttribute('data-open-history')||'replace', updateHistory:true});
        return;
      }
      if(t.hasAttribute('data-filter')){
        state.filter=t.getAttribute('data-filter')||'all';
        state.page=0;
        // «Недавно добавленные» / «Новинки» — нужны все чанки (игры часто в поздних)
        if((state.filter==='recent'||state.filter==='new') && state.manifest && !state.allChunksLoaded && !state.allChunksLoading){
          state.allChunksLoading=true;
          const want=state.filter;
          loadAllChunksBackground(state.manifest).then(()=>{
            if(state.filter===want){
              $('[data-filters]',mount).innerHTML=filterChips();
              renderGrid();
            }
          });
        }
        if(state.filter==='new'){
          ensureNovinki().then(()=>{
            if(state.filter==='new'){
              $('[data-filters]',mount).innerHTML=filterChips();
              renderGrid();
            }
          });
        }
        $('[data-filters]',mount).innerHTML=filterChips();
        renderGrid();
        return;
      }
      if(t.hasAttribute('data-theme-toggle')){
        state.theme=state.theme==='dark'?'light':'dark';
        try{ localStorage.setItem(THEME_KEY,state.theme); }catch(e){}
        applyTheme(); return;
      }
      if(t.hasAttribute('data-close-detail')){
        // Кнопка "← Назад" внутри карточки:
        // если карточка была открыта через рекомендацию (history.pushState),
        // то возвращаемся в историю, иначе закрываем в главное.
        try{
          const hs=history && history.state ? history.state : {};
          if(state.open && hs && hs.rentCard && hs.pushed){
            history.back();
            return;
          }
        }catch(e){}
        closeDetail();
        return;
      }
      if(t.hasAttribute('data-sticky-buy')){ addToCart(false); return; }
      if(t.hasAttribute('data-sticky-cart')){ addToCart(true); return; }
      if(t.hasAttribute('data-sticky-share')){ share(); return; }
      if(t.hasAttribute('data-close-gallery')){ closeGallery(); return; }
      if(t.hasAttribute('data-gallery-prev')){ stepShot(-1); return; }
      if(t.hasAttribute('data-gallery-next')){ stepShot(1); return; }
      if(t.hasAttribute('data-page')){
        if(t.disabled) return;
        const next=Math.max(0, +t.getAttribute('data-page')||0);
        state.page=next;
        renderGrid();
        scrollCatalogToTop();
        return;
      }
      if(t.hasAttribute('data-fab-up')){
        if(state.open){
          const sc=$('.rt-detail__scroll',mount);
          sc?.scrollTo({top:0, behavior:'smooth'});
        }else{
          try{ window.scrollTo({top:0, behavior:'smooth'}); }catch(e){ window.scrollTo(0,0); }
        }
        return;
      }
    });
    function syncSearchClear(){
      const has=!!String(state.query||'').trim();
      $$('[data-search-clear]',mount).forEach(btn=>{
        btn.classList.toggle('is-on', has);
      });
    }
    $$('[data-top-search]',mount).forEach(inp=>{
      inp.addEventListener('input', debounce(()=>{
        const q=inp.value||'';
        state.query=q;
        state.page=0;
        $$('[data-top-search]',mount).forEach(x=>{ if(x!==inp) x.value=state.query; });
        syncSearchClear();
        if(state.open) closeDetail();
        $('[data-filters]',mount).innerHTML=filterChips();
        const grid=$('[data-grid]',mount);
        if(grid && String(q).trim() && !filteredItems().length){
          grid.innerHTML='<div class="rt-status">Поиск…</div>';
        }
        prefetchSearchChunks(q).finally(()=>{
          if(state.query!==q) return;
          renderGrid();
        });
      },160));
    });
    syncSearchClear();
    document.addEventListener('keydown', e=>{
      const tag=(e.target&&e.target.tagName||'').toLowerCase();
      if(tag==='input'||tag==='textarea'||(e.target&&e.target.isContentEditable)) return;
      if(e.key==='Escape'){
        if(isGalleryOpen()){ closeGallery(); return; }
        if(state.open) closeDetail();
        return;
      }
      if(e.key==='ArrowLeft'||e.key==='ArrowRight'){
        if(!state.open) return;
        if(e.key==='ArrowLeft'){ if(stepShot(-1)) e.preventDefault(); }
        else { if(stepShot(1)) e.preventDefault(); }
      }
    });
    const galleryModal=$('#rtGallery',mount);
    bindGallerySwipe(galleryModal);
    galleryModal?.addEventListener('click', e=>{
      if(e.target===galleryModal) closeGallery();
    });
    window.addEventListener('hashchange', ()=>{
      const m=(location.hash||'').match(HASH_RE);
      if(m) openDetailBySlug(decodeURIComponent(m[1]), {updateHistory:false});
      else if(state.open) closeDetail();
    });

    // Floating "back to top" button (main page + inside opened card)
    const fab=$('[data-fab-up]',mount);
    if(fab){
      const upd=debounce(()=>{
        const y = state.open ? ($('.rt-detail__scroll',mount)?.scrollTop||0) : (window.scrollY||0);
        if(y>700) fab.classList.add('is-show');
        else fab.classList.remove('is-show');
      }, 120);
      window.addEventListener('scroll', upd, {passive:true});
      const sc=$('.rt-detail__scroll',mount);
      sc?.addEventListener('scroll', upd, {passive:true});
      upd();
    }
  }

  (async function init(){
    shell();
    bindUI();
    applyMobileGuard();
    $('[data-filters]',mount).innerHTML=filterChips();
    try{
      const manifest=await loadManifest();
      state.manifest=manifest;
      const novinkiPromise=ensureNovinki();
      await loadSearchIndex();
      ensureRecsPriority();
      await novinkiPromise;
      const chunks=(manifest.chunks||[]);
      const hashSlug=((location.hash||'').match(HASH_RE)||[])[1];
      const prioritySlug=hashSlug?decodeURIComponent(hashSlug):'';
      let priorityIdx=null;
      if(prioritySlug && state.searchIndex){
        const hit=state.searchIndex.find(x=>x&&x.slug===prioritySlug);
        if(hit) priorityIdx=hit.chunk;
      }
      if(chunks.length){
        if(priorityIdx!=null && +priorityIdx>0){
          await loadChunk(chunks[0],0);
          await loadChunkByIndex(+priorityIdx);
        }else{
          await loadChunk(chunks[0],0);
        }
        // Первая страница (chunk_000) — сразу на экран
        state.firstPageReady=true;
        $('[data-filters]',mount).innerHTML=filterChips();
        renderGrid();
        // Если на странице больше одного чанка (104) — догрузим без зависания
        if(pageSize()>((+manifest.chunk_size)||100)*0.9){
          ensureChunksForPage(0).then(()=>{ if(!state.open) renderGrid(); });
        }
      } else if(manifest.items){
        mergeItems(manifest.items);
        state.firstPageReady=true;
      }
      // Deep-link сразу
      const m=(location.hash||'').match(HASH_RE);
      if(m) openDetailBySlug(decodeURIComponent(m[1]), {updateHistory:false});
      if(chunks.length<=1){
        state.allChunksLoaded=true;
        state.firstPageReady=true;
        clearCatalogLoadingTip();
        $('[data-filters]',mount).innerHTML=filterChips();
        renderGrid();
      } else {
        state.allChunksLoading=true;
        setCatalogLoadingStatus(state.chunksLoaded.size, chunks.length);
        loadAllChunksBackground(manifest, {fromIdx:1}); // остальное в фоне
      }
    }catch(err){
      try{
        const data=await fetchJson([CDN+'rent-catalog/rent_catalog.json', './catalog/rent_catalog.json']);
        mergeItems(Array.isArray(data)?data:(data.items||[]));
        state.allChunksLoaded=true;
        state.firstPageReady=true;
        invalidateSpotlight();
        renderGrid();
      }catch(e2){
        $('[data-grid]',mount).innerHTML=`<div class="rt-status">Ошибка загрузки каталога: ${esc((err&&err.message)||err)}</div>`;
      }
    }
  })();
})();
